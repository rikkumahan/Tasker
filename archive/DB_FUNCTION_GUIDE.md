## 0. Prerequisites

### Schema additions (run before creating any function)
```sql
-- Enable extensions
create extension if not exists vector;

-- Core tables with user_id
create table if not exists emails (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id),
  message_id  text not null,
  thread_id   text,                        -- Gmail thread ID
  sender      text not null,
  subject     text,
  summary     text not null,
  body        text,
  embedding   vector(1536) not null,       -- embedded from summary only
  deep_link   text,
  direction   text default 'inbox',        -- 'inbox' | 'sent'
  created_at  timestamptz default now(),
  unique(user_id, message_id)
);

create table if not exists entities (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id),
  name        text not null,
  type        text not null,               -- PERSON | ORGANIZATION | PROJECT | DEADLINE | TOPIC | LOCATION
  embedding   vector(1536),
  created_at  timestamptz default now()
);

create table if not exists relationships (
  id           bigint generated by default as identity primary key,
  user_id      uuid not null references auth.users(id),
  source_id    uuid references emails(id) on delete cascade,
  target_id    uuid references entities(id) on delete cascade,
  relation_type text not null,             -- MENTIONS | SENT_BY | SENT_TO | ASSIGNED_TO | RELATED_TO | HAS_DEADLINE | DELEGATED_TO | COMMITTED_TO
  weight       float default 1.0,          -- PPR edge weight
  created_at   timestamptz default now()
);

create table if not exists action_items (
  id           uuid primary key default gen_random_uuid(),
  user_id      uuid not null references auth.users(id),
  email_id     uuid references emails(id),
  description  text not null,
  status       text default 'pending',     -- 'pending' | 'done' | 'duplicate'
  embedding    vector(1536) not null,
  assigned_to  text,
  deadline     date,
  direction    text,                       -- 'inbox' (others committed to you) | 'sent' (you committed)
  created_at   timestamptz default now()
);

create table if not exists email_entities (
  id        uuid primary key default gen_random_uuid(),
  user_id   uuid not null references auth.users(id),
  email_id  uuid references emails(id),
  value     text not null,
  type      text not null,
  role      text
);

-- HNSW indexes for vector search
create index if not exists emails_embedding_idx    on emails    using hnsw (embedding vector_cosine_ops);
create index if not exists entities_embedding_idx  on entities  using hnsw (embedding vector_cosine_ops);
create index if not exists actions_embedding_idx   on action_items using hnsw (embedding vector_cosine_ops);

-- Performance indexes
create index if not exists emails_user_idx         on emails(user_id, created_at desc);
create index if not exists emails_sender_idx       on emails(user_id, sender);
create index if not exists emails_thread_idx       on emails(user_id, thread_id);
create index if not exists relationships_source_idx on relationships(source_id);
create index if not exists relationships_target_idx on relationships(target_id);
create index if not exists action_items_user_idx   on action_items(user_id, status, deadline);
```

## Function 1: `match_emails` — Vector Similarity Retrieval

**What:** Finds past emails semantically similar to a query embedding using pgvector cosine similarity.

**When called:**
- Ingestion: pre-fetch relevant history before LLM extracts actions
- Query agent: find relevant emails for a user's natural language question

**Why it matters:** This is your token efficiency core. Instead of injecting all 50 emails into the LLM, you inject only the top-K most semantically relevant ones.

```sql
create or replace function match_emails(
  query_embedding  vector(1536),
  p_user_id        uuid,
  match_threshold  float default 0.75,
  match_count      int   default 8
)
returns table (
  id          uuid,
  sender      text,
  subject     text,
  summary     text,
  similarity  float,
  sent_date   date,
  direction   text
)
language plpgsql as $$
begin
  return query
  select
    e.id,
    e.sender,
    e.subject,
    e.summary,
    1 - (e.embedding <=> query_embedding) as similarity,
    e.created_at::date,
    e.direction
  from emails e
  where e.user_id = p_user_id
    and 1 - (e.embedding <=> query_embedding) > match_threshold
  order by similarity desc
  limit match_count;
end;
$$;
```

**Edge Function call:**
```typescript
const { data: context } = await supabase.rpc("match_emails", {
  query_embedding:  emailEmbedding,
  p_user_id:        userId,
  match_threshold:  0.75,
  match_count:      8
});
// Inject context.map(c => c.summary) into LLM system prompt
```

## Function 2: `get_email_entities` — 1-Hop Graph Traversal

**What:** Returns all entities directly linked to a specific email via the relationships table.

**When called:**
- After email insert: verify what was linked
- Build context block for LLM
- Email detail view in UI

**Why it matters:** 1-hop only = single join, no explosion risk. Tells you exactly what this email is about structurally.

```sql
create or replace function get_email_entities(
  p_email_id  uuid,
  p_user_id   uuid
)
returns table (
  entity_id    uuid,
  entity_name  text,
  entity_type  text,
  relation     text,
  weight       float
)
language sql as $$
  select
    en.id,
    en.name,
    en.type,
    r.relation_type,
    r.weight
  from entities en
  join relationships r on r.target_id = en.id
  where r.source_id = p_email_id
    and r.user_id   = p_user_id;
$$;
```

**Edge Function call:**
```typescript
const { data: entities } = await supabase.rpc("get_email_entities", {
  p_email_id: emailId,
  p_user_id:  userId
});
```

## Function 3: `get_related_emails` — PPR-Based Traversal

**What:** Finds emails related to a given email by traversing shared entities, scoring each hop with a PPR decay factor (0.85). Cuts off below threshold — prevents hub node explosion.

**When called:**
- Related Mail Clustering feature
- Context enrichment during ingestion (alongside `match_emails`)

**Why PPR over 2-hop:** Fixed 2-hop returns ALL emails sharing an entity — if "Meeting" is in 40 emails, you get 40 rows. PPR scores decay per hop so irrelevant results drop below threshold and are excluded.

```sql
create or replace function get_related_emails(
  p_email_id     uuid,
  p_user_id      uuid,
  decay_factor   float default 0.85,
  min_score      float default 0.3,
  max_results    int   default 10
)
returns table (
  email_id      uuid,
  sender        text,
  subject       text,
  summary       text,
  sent_date     date,
  ppr_score     float,
  shared_entity text
)
language plpgsql as $$
begin
  return query
  with seed_entities as (
    -- Get entities from source email (hop 0)
    select r.target_id as entity_id, r.weight as edge_weight
    from relationships r
    where r.source_id = p_email_id
      and r.user_id   = p_user_id
  ),
  scored_emails as (
    -- Hop 1: emails sharing same entities, scored by PPR decay
    select
      e.id,
      e.sender,
      e.subject,
      e.summary,
      e.created_at::date,
      -- PPR score = edge_weight × decay_factor × relationship_weight
      (se.edge_weight * decay_factor * r2.weight) as score,
      en.name as entity_name
    from seed_entities se
    join relationships r2  on r2.target_id = se.entity_id
    join emails e           on e.id         = r2.source_id
    join entities en        on en.id        = se.entity_id
    where e.id      != p_email_id
      and e.user_id  = p_user_id
      and (se.edge_weight * decay_factor * r2.weight) >= min_score
  )
  select
    id,
    sender,
    subject,
    summary,
    created_at,
    max(score)       as ppr_score,
    max(entity_name) as shared_entity
  from scored_emails
  group by id, sender, subject, summary, created_at
  order by ppr_score desc
  limit max_results;
end;
$$;
```

**Edge Function call:**
```typescript
const { data: related } = await supabase.rpc("get_related_emails", {
  p_email_id:   emailId,
  p_user_id:    userId,
  decay_factor: 0.85,
  min_score:    0.3,
  max_results:  10
});
```

## Function 4: `check_duplicate_task` — Action Item Deduplication

**What:** Before inserting a new action item, checks if a semantically identical pending task exists using vector cosine similarity at 0.92 threshold.

**When called:** Every time a new action item is about to be inserted — during ingestion.

**Why it matters:** Prevents "Send report to John" and "Please send the report to John" from becoming two separate tasks across 50 emails.

```sql
create or replace function check_duplicate_task(
  task_vector    vector(1536),
  p_user_id      uuid,
  sim_threshold  float default 0.92
)
returns table (
  id             uuid,
  original_task  text,
  similarity     float
)
language plpgsql as $$
begin
  return query
  select
    a.id,
    a.description,
    1 - (a.embedding <=> task_vector) as similarity
  from action_items a
  where a.user_id = p_user_id
    and a.status  = 'pending'
    and 1 - (a.embedding <=> task_vector) > sim_threshold
  order by similarity desc
  limit 1;
end;
$$;
```

**Edge Function call:**
```typescript
const taskEmbedding = await embed(actionDescription);

const { data: dupes } = await supabase.rpc("check_duplicate_task", {
  task_vector:   taskEmbedding,
  p_user_id:     userId,
  sim_threshold: 0.92
});

await supabase.from("action_items").insert({
  user_id:     userId,
  email_id:    emailId,
  description: actionDescription,
  status:      dupes?.length ? "duplicate" : "pending",
  embedding:   taskEmbedding,
  deadline:    action.deadline ?? null,
  assigned_to: action.assigned_to ?? null,
  direction:   emailDirection
});
```

## Function 5: `get_pending_deadlines` — Deadline Tracker

**What:** Returns all pending action items with deadlines, ordered by urgency. Covers both commitments made TO the user (inbox) and BY the user (sent).

**When called:** Dashboard load, notification trigger, Deadline Tracker feature view.

```sql
create or replace function get_pending_deadlines(p_user_id uuid)
returns table (
  action_id       uuid,
  description     text,
  deadline        date,
  days_remaining  int,
  assigned_to     text,
  direction       text,
  email_subject   text,
  deep_link       text,
  is_overdue      boolean
)
language sql as $$
  select
    a.id,
    a.description,
    a.deadline,
    (a.deadline - current_date)::int as days_remaining,
    a.assigned_to,
    a.direction,
    e.subject,
    e.deep_link,
    (a.deadline < current_date) as is_overdue
  from action_items a
  join emails e on a.email_id = e.id
  where a.user_id = p_user_id
    and a.status  = 'pending'
    and a.deadline is not null
  order by a.deadline asc;
$$;
```

**Edge Function call:**
```typescript
const { data: deadlines } = await supabase.rpc("get_pending_deadlines", {
  p_user_id: userId
});
// Split: deadlines.filter(d => d.is_overdue) → overdue section
//        deadlines.filter(d => !d.is_overdue) → upcoming section
```

## Function 6: `get_action_insights` — Action Insights Feature

**What:** Returns all pending action items grouped by entity (project/person/topic) with counts. Powers the Action Insights dashboard view.

**When called:** Main Action Insights view, workload overview.

```sql
create or replace function get_action_insights(p_user_id uuid)
returns table (
  entity_name  text,
  entity_type  text,
  action_count bigint,
  actions      json
)
language sql as $$
  select
    en.name,
    en.type,
    count(a.id),
    json_agg(json_build_object(
      'id',          a.id,
      'description', a.description,
      'deadline',    a.deadline,
      'assigned_to', a.assigned_to,
      'direction',   a.direction,
      'status',      a.status
    ) order by a.deadline asc nulls last) as actions
  from action_items a
  join emails e       on a.email_id   = e.id
  join relationships r on r.source_id = e.id
  join entities en    on r.target_id  = en.id
  where a.user_id = p_user_id
    and a.status  = 'pending'
  group by en.name, en.type
  order by count(a.id) desc;
$$;
```

**Edge Function call:**
```typescript
const { data: insights } = await supabase.rpc("get_action_insights", {
  p_user_id: userId
});
// insights = [{ entity_name: "Project Titan", action_count: 5, actions: [...] }]
```

## Function 9: `get_entity_timeline` — Entity Evolution Over Time

**What:** Returns all emails mentioning a specific entity, ordered by date — shows how a project/person/topic evolved over time.

**When called:** Entity detail view in UI, timeline feature.

```sql
create or replace function get_entity_timeline(
  p_entity_name  text,
  p_user_id      uuid,
  from_date      date default null,
  to_date        date default null
)
returns table (
  email_id      uuid,
  sender        text,
  subject       text,
  summary       text,
  sent_date     date,
  relation_type text,
  direction     text
)

    e.id,
    e.sender,
    e.subject,
    e.summary,
    e.created_at::date,
    r.relation_type,
    e.direction
  from emails e
  join relationships r on r.source_id = e.id
  join entities en     on r.target_id = en.id
  where en.name     = p_entity_name
    and e.user_id   = p_user_id
    and (from_date is null or e.created_at::date >= from_date)
    and (to_date   is null or e.created_at::date <= to_date)
  order by e.created_at asc;
$$;
```

## Function 12: `get_delegation_chain` — Delegation Tracker

**What:** Returns all tasks delegated TO others (sent direction, DELEGATED_TO relation), with their current status. Uses PPR scoring to surface most relevant chains.

**When called:** "What have I delegated?" view, follow-up tracker.

```sql
create or replace function get_delegation_chain(
  p_user_id     uuid,
  decay_factor  float default 0.85,
  min_score     float default 0.3
)
returns table (
  action_id    uuid,
  description  text,
  delegated_to text,
  deadline     date,
  status       text,
  email_subject text,
  deep_link    text,
  ppr_score    float
)
language plpgsql as $$
begin
  return query
  with delegated as (
    select
      a.id,
      a.description,
      a.assigned_to,
      a.deadline,
      a.status,
      e.subject,
      e.deep_link,
      -- PPR score based on relationship weight and decay
      (r.weight * decay_factor) as score
    from action_items a
    join emails e       on a.email_id   = e.id
    join relationships r on r.source_id = e.id
    where a.user_id    = p_user_id
      and a.direction  = 'sent'
      and r.relation_type = 'DELEGATED_TO'
      and (r.weight * decay_factor) >= min_score
  )
  select
    id,
    description,
    assigned_to,
    deadline,
    status,
    subject,
    deep_link,
    max(score) as ppr_score
  from delegated
  group by id, description, assigned_to, deadline, status, subject, deep_link
  order by ppr_score desc, deadline asc nulls last;
end;
$$;
```

**Edge Function call:**
```typescript
const { data: delegations } = await supabase.rpc("get_delegation_chain", {
  p_user_id:    userId,
  decay_factor: 0.85,
  min_score:    0.3
});
```

## Full Ingestion Pipeline (How All Functions Wire Together)

```typescript
async function processEmail(gmailMsg: any, userId: string) {

  const { sender, subject, body, threadId, messageId, direction } = parseGmail(gmailMsg);

  // ── Step 1: LLM — extract structure (ONE call, context injected)
  const { summary, entities, actions } = await analyzeEmail(sender, subject, body);

  // ── Step 2: Embed summary
  const emailEmbedding = await embed(summary);

  // ── Step 3: Pre-fetch ALL context in parallel (no LLM waiting)
  const [vectorContext, threadContext, senderContext] = await Promise.all([
    supabase.rpc("match_emails",      { query_embedding: emailEmbedding, p_user_id: userId }),
    supabase.rpc("get_thread_context",{ p_thread_id: threadId, p_user_id: userId }),
    supabase.rpc("get_sender_context",{ p_sender: sender, p_user_id: userId })
  ]);

  // ── Step 4: Insert email record
  const { data: emailRow } = await supabase.from("emails").insert({
    user_id: userId, message_id: messageId, thread_id: threadId,
    sender, subject, summary, body, embedding: emailEmbedding, direction
  }).select("id").single();

  // ── Step 5: Insert entities + relationships (from your existing extraction)
  for (const entity of entities) {
    const entityId = await resolveOrCreateEntity(entity, userId);
    await supabase.from("relationships").insert({
      user_id: userId, source_id: emailRow.id,
      target_id: entityId, relation_type: entity.relation, weight: 1.0
    });
  }

  // ── Step 6: Fetch graph context per entity (PPR)
  const graphContext = await supabase.rpc("get_related_emails", {
    p_email_id: emailRow.id, p_user_id: userId
  });

  // ── Step 7: Re-run LLM with FULL context for action extraction
  const enrichedActions = await extractActionsWithContext({
    emailBody:     body,
    vectorContext: vectorContext.data,
    threadContext: threadContext.data,
    senderContext: senderContext.data,
    graphContext:  graphContext.data
  });

  // ── Step 8: Write action items with dedup
  for (const action of enrichedActions) {
    const taskEmbedding = await embed(action.description);
    const { data: dupes } = await supabase.rpc("check_duplicate_task", {
      task_vector: taskEmbedding, p_user_id: userId
    });
    await supabase.from("action_items").insert({
      user_id:     userId,
      email_id:    emailRow.id,
      description: action.description,
      status:      dupes?.length ? "duplicate" : "pending",
      embedding:   taskEmbedding,
      deadline:    action.deadline ?? null,
      assigned_to: action.assigned_to ?? null,
      direction
    });
  }
}
```

## Query Agent Pipeline (Read Path)

```typescript
async function queryAgent(query: string, userId: string) {

  // 1. Embed the query
  const qEmbedding = await embed(query);

  // 2. Fetch context in parallel
  const [emailContext, taskContext] = await Promise.all([
    supabase.rpc("match_emails",       { query_embedding: qEmbedding, p_user_id: userId }),
    supabase.rpc("match_action_items", { query_embedding: qEmbedding, p_user_id: userId })
  ]);

  // 3. Build context block (summaries ONLY — never body)
  const contextBlock = [
    ...emailContext.data.map(c => `[${c.sent_date}] ${c.sender}: ${c.summary}`),
    ...taskContext.data.map(t => `[TASK] ${t.description} (${t.status})`)
  ].join("\n");

  // 4. LLM answers grounded in context
  const answer = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: `Answer using ONLY this context:\n${contextBlock}` },
      { role: "user",   content: query }
    ]
  });

  return {
    answer:   answer.choices[0].message.content,
    sources:  emailContext.data.map(c => ({ date: c.sent_date, summary: c.summary }))
  };
}
```
